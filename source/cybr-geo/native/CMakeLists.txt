cmake_minimum_required(VERSION 3.18)
project(mechanism_pathtrace LANGUAGES CXX)
find_package(OpenMP REQUIRED)

add_executable(mechanism_pathtrace pathtrace.cpp)
target_compile_features(mechanism_pathtrace PRIVATE cxx_std_17)
target_compile_options(mechanism_pathtrace PRIVATE $<$<CXX_COMPILER_ID:GNU,Clang>:-O3>)
target_link_libraries(mechanism_pathtrace PRIVATE OpenMP::OpenMP_CXX)

add_executable(mechanism_photoreal photoreal.cpp)
target_compile_features(mechanism_photoreal PRIVATE cxx_std_17)
target_compile_options(mechanism_photoreal PRIVATE $<$<CXX_COMPILER_ID:GNU,Clang>:-O3>)
target_link_libraries(mechanism_photoreal PRIVATE OpenMP::OpenMP_CXX)

add_executable(mechanism_photographic_volume photographic_volume.cpp)
target_compile_features(mechanism_photographic_volume PRIVATE cxx_std_17)
target_compile_options(mechanism_photographic_volume PRIVATE $<$<CXX_COMPILER_ID:GNU,Clang>:-O3>)
target_link_libraries(mechanism_photographic_volume PRIVATE OpenMP::OpenMP_CXX)

include(CTest)
if(BUILD_TESTING)
  add_executable(test_photographic_volume test_photographic_volume.cpp)
  target_compile_features(test_photographic_volume PRIVATE cxx_std_17)
  target_compile_options(test_photographic_volume PRIVATE $<$<CXX_COMPILER_ID:GNU,Clang>:-O3>)
  target_link_libraries(test_photographic_volume PRIVATE OpenMP::OpenMP_CXX)
  add_test(NAME photographic_volume_analytic COMMAND test_photographic_volume)
endif()

  foreach(test_name IN ITEMS test_legacy_water_connection test_spectral_desert_v5 test_broadband_ripples test_water_medium test_photographic_grain test_water_microfacet test_reflected_connection test_cloud_volume_v9 test_reflectance_calibration_v9)

    target_compile_features(${test_name} PRIVATE cxx_std_17)
    target_link_libraries(${test_name} PRIVATE OpenMP::OpenMP_CXX)
    if(test_name STREQUAL "test_photographic_grain")

      add_test(NAME ${test_name} COMMAND ${test_name})
    endif()
  endforeach()
  add_test(NAME spectral_desert_optical_smoke COMMAND cybr_spectral_desert --self-test)
